    }

# [PUBLISHED] 4,879 -> 1,963 -> 360 -> 158; Abstract, Results, Figure 2B
query ImproveAnalysis {
  # Verified count; `python scripts/verify_counts.py verify` re-checks it.
  # expect NonG2A_Missense: 4879
  # expect NonG2A_Missense_Editable: 1963
  # expect NonG2A_Missense_Improve: 158
  # expect NonG2A_Missense_Improve_Simple: 360
  # Improve (neighbor-drill) funnel — non-G>A missense in the SFARI protein-coding universe.
  # Reported entirely on the SIFT scale (a missense carries its own SIFT; no CADD axis, unlike Rescue).

  # 1) Missense pool (Improve candidates), analogous to NonG2A_StopGained above.
  NonG2A_Missense: variants_aggregate(
    where: {
      class: {_eq: "SNV"},
      _not: {
        _or: [
          {ref: {_eq: "G"}, alt: {_eq: "A"}, gene: {strand: {_eq: "+"}}},
          {ref: {_eq: "C"}, alt: {_eq: "T"}, gene: {strand: {_eq: "-"}}}
        ]
      },
      variants_features: {feature: {biotype: {name: {_eq: "protein_coding"}}}},
      gene: {genes_quantitative_scores: {quantitative_score: {name: {_eq: "SFARI Gene Score"}}}},
      variants_consequences: {consequence: {name: {_eq: "Missense"}}}
    }
  ) {
    aggregate {
      count
    }
  }

  # 2) ADAR-addressable — the codon contains at least one editable adenosine (ADAR edits A->I only;
  #    a missense whose codon has no A cannot be targeted by this strategy at all).
  NonG2A_Missense_Editable: variants_aggregate(
    where: {
      class: {_eq: "SNV"},
      _not: {
        _or: [
          {ref: {_eq: "G"}, alt: {_eq: "A"}, gene: {strand: {_eq: "+"}}},
          {ref: {_eq: "C"}, alt: {_eq: "T"}, gene: {strand: {_eq: "-"}}}
        ]
      },
      variants_features: {feature: {biotype: {name: {_eq: "protein_coding"}}}},
      gene: {genes_quantitative_scores: {quantitative_score: {name: {_eq: "SFARI Gene Score"}}}},
      variants_consequences: {consequence: {name: {_eq: "Missense"}}},
      # ── codon has >=1 adenosine to edit (the neighbor-drill candidate filter) ──
      alt_codon: {_or: [{nt1: {_eq: "A"}}, {nt2: {_eq: "A"}}, {nt3: {_eq: "A"}}]}
    }
  ) {
    aggregate {
      count
    }
  }

  # 3) "Simple" improves — the best edit is tolerated & better than the variant AA, but this STILL
  #    counts variants whose OWN residue was already tolerated (improvement present, yet not that
  #    interesting clinically).
  NonG2A_Missense_Improve_Simple: variants_aggregate(
    where: {
      class: {_eq: "SNV"},
      _not: {
        _or: [
          {ref: {_eq: "G"}, alt: {_eq: "A"}, gene: {strand: {_eq: "+"}}},
          {ref: {_eq: "C"}, alt: {_eq: "T"}, gene: {strand: {_eq: "-"}}}
        ]
      },
      variants_features: {feature: {biotype: {name: {_eq: "protein_coding"}}}},
      gene: {genes_quantitative_scores: {quantitative_score: {name: {_eq: "SFARI Gene Score"}}}},
      variants_consequences: {consequence: {name: {_eq: "Missense"}}},
      neighbor_edits: {is_best: {_eq: true}, improves: {_eq: true}}
    }
  ) {
    aggregate {
      count
    }
  }

  # 4) Really improved — a DELETERIOUS residue (variant's own SIFT < 0.05) turned TOLERATED by the
  #    best edit (SIFT >= 0.05). This is the reportable headline (deleterious -> tolerated).
  NonG2A_Missense_Improve: variants_aggregate(
    where: {
      class: {_eq: "SNV"},
      _not: {
        _or: [
          {ref: {_eq: "G"}, alt: {_eq: "A"}, gene: {strand: {_eq: "+"}}},
          {ref: {_eq: "C"}, alt: {_eq: "T"}, gene: {strand: {_eq: "-"}}}
        ]
      },
      variants_features: {feature: {biotype: {name: {_eq: "protein_coding"}}}},
      gene: {genes_quantitative_scores: {quantitative_score: {name: {_eq: "SFARI Gene Score"}}}},
      variants_consequences: {consequence: {name: {_eq: "Missense"}}},
      # ── variant is deleterious: its own SIFT < 0.05 ──
      variants_quantitative_scores: {quantitative_score: {name: {_eq: "SIFT"}, value: {_lt: 0.05}}},
      # ── best neighbor (Improve) edit is tolerated (SIFT >= 0.05) & better than the variant AA ──
      neighbor_edits: {is_best: {_eq: true}, improves: {_eq: true}}
    }
  ) {
    aggregate {
      count
    }
  }
}

query TreatableVariants($limit: Int!, $offset: Int!) {
  TreatableVariants: variants(
    where: {
      class: {_eq: "SNV"}
      _or: [
        {ref: {_eq: "G"}, alt: {_eq: "A"}, gene: {strand: {_eq: "+"}}}
        {ref: {_eq: "C"}, alt: {_eq: "T"}, gene: {strand: {_eq: "-"}}}
      ]
      variants_features: {feature: {biotype: {name: {_eq: "protein_coding"}}}}
      gene: {genes_quantitative_scores: {quantitative_score: {name: {_eq: "SFARI Gene Score"}}}}
      variants_consequences: {consequence: {name: {_in: ["StopGained", "Missense", "Splice"]}}}
    }
    order_by: {id: asc}
    limit: $limit
    offset: $offset
  ) {
    id
    coordinate { chr start end }
    ref
    alt
    ref_codon { nt1 nt2 nt3 }
    alt_codon { nt1 nt2 nt3 }
    variants_consequences { consequence { name } }
    gene {
      ensg
      symbol
      genes_quantitative_scores(where: {quantitative_score: {name: {_eq: "SFARI Gene Score"}}}) {
        quantitative_score { value }
      }
      genes_qualitative_scores(where: {qualitative_score: {name: {_eq: "SFARI Syndromic"}}}) {
        qualitative_score { value }
      }
    }
    cds_position
    variants_quantitative_scores(where: {quantitative_score: {name: {_in: ["SIFT", "CADD"]}}}) {
      quantitative_score { name value }
    }
    variants_guides(where: {guide: {type: {_eq: "pre_mRNA"}}}) {
      guide { hits_85 }
      bystanders {
        id
        genomic_position
        bystanders_consequences { consequence { name } }
        cds_position
        in_cds
        sift_score
        sift_prediction
        cadd_phred
        ref_codon { nt1 nt2 nt3 amino_acid { short_name } }
        alt_codon { nt1 nt2 nt3 amino_acid { short_name } }
      }
    }
  }
}
